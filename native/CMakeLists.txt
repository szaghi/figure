# FIGURE native layer: Dear ImGui + ImPlot + ImPlot3D + cimgui + cimplot + cimplot3d + SDL3/OpenGL3 backends + the vendored
# extensions (ImGuiColorTextEdit, ImGuizmo, imgui_rich_md with md4c) + FIGURE glue.
#
# Built once, before any Fortran build, by scripts/build_native.sh. Neither FoBiS nor fpm compiles C++: both only link
# the static libraries this project installs into <prefix>/lib (default: the repository's lib/).
#
# Pinned third-party versions:
#   cimgui                -> Dear ImGui 1.93.1 WIP, docking branch at cf53467 (2026-11-10), vendored in
#                            third_party/cimgui: cimgui has no tag there, so FIGURE runs cimgui's generator (its tag
#                            1.93.1WIPdock) with scripts/gen_cimgui.sh (LuaJIT; to move Dear ImGui only, not to build)
#   cimplot v1.1          -> ImPlot 1.2 WIP (git submodule of cimplot); generated against cimgui 1.93dock, the same
#                            Dear ImGui 1.93 API (docking only adds to it)
#   cimplot3d v0.5        -> the C API of ImPlot3D 0.6 WIP and its generator metadata (read by gen_bindings.py),
#                            generated against the same Dear ImGui 1.93 API; fetched without its submodule
#   ImPlot3D              -> the small 3-D figures (ip3_*; MIT): brenocq/implot3d at commit c54525f (the submodule
#                            cimplot3d v0.5 pins: v0.5 with its version bumped), vendored in third_party/implot3d with
#                            FIGURE's one patch (its patches/ directory). It needs Dear ImGui only (not ImPlot) and
#                            draws through its draw lists
#   SDL     release-3.4.20
#   miniz   3.0.2         -> deflate for the remote mode's compressed messages (MIT)
#   stb_image_write v1.16 -> JPEG for the remote mode's changing images, PNG and JPEG files (MIT or public domain),
#                            vendored unmodified in third_party/stb (one header)
#   stb_image v2.30       -> PNG and JPEG files read (fig_read_image, tex%load; MIT or public domain), vendored
#                            unmodified in third_party/stb (one header), the same stb commit as stb_image_write

#                            the goossens fork at commit 9f812a0, vendored in third_party/ImGuiColorTextEdit with

# ---- FIGURE's patches to the fetched Dear ImGui (native/patches/imgui-*.patch, THIRD_PARTY.md) ---------------------
# Each patch changes one line. It is applied to a copy of the file in the build directory, compiled instead of the
# fetched one (the fetched sources stay as fetched, a FETCHCONTENT_SOURCE_DIR_CIMGUI included); scripts/fpm_package.py
# applies the same patches to the fpm package. A patch whose line is no longer in the file stops the configuration:
# upstream changed it (fixed, perhaps), and the patch is checked against its upstream issue, then updated or removed.
function(figure_patch_line source patch out)
   file(READ ${patch} diff)
   string(REPLACE "\r" "" diff "${diff}") # a checkout with CRLF line ends
   string(REGEX MATCH "\n-([^-\n][^\n]*)" old "${diff}") # the removed line, not the --- header
   set(old "${CMAKE_MATCH_1}")
   string(REGEX MATCH "\n\\+([^+\n][^\n]*)" new "${diff}") # the added line, not the +++ header
   set(new "${CMAKE_MATCH_1}")
   file(READ ${source} text)
   string(FIND "${text}" "${old}" at)
   if(at EQUAL -1)
      message(FATAL_ERROR "${patch} does not apply to ${source}: its line changed upstream. Check the patch against "
                          "its upstream issue (THIRD_PARTY.md), then update or remove it.")
   endif()
   string(REPLACE "${old}" "${new}" text "${text}")
   if(EXISTS ${out})
      file(READ ${out} current)
   endif()
   if(NOT "${current}" STREQUAL "${text}")
      file(WRITE ${out} "${text}")
   endif()
   set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${patch} ${source})
endfunction()

# themes/looks/<name>.toml and themes/palettes/<name>.toml (reconfigured when one changes).

   if(theme_kind STREQUAL "THEMES")
      set(theme_dir ${CMAKE_CURRENT_SOURCE_DIR}/../themes)
   elseif(theme_kind STREQUAL "LOOKS")
      set(theme_dir ${CMAKE_CURRENT_SOURCE_DIR}/../themes/looks)

      set(theme_dir ${CMAKE_CURRENT_SOURCE_DIR}/../themes/palettes)
   endif()

   list(SORT FIGURE_THEME_FILES)
   set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${FIGURE_THEME_FILES} ${theme_dir})
   string(APPEND FIGURE_THEMES_INC "const theme_source ${theme_kind}[] = {\n")
   foreach(theme_file ${FIGURE_THEME_FILES})
      get_filename_component(theme_name ${theme_file} NAME_WE)
      file(READ ${theme_file} theme_text)
      string(APPEND FIGURE_THEMES_INC "   {\"${theme_name}\", R\"FIGURETHEME(${theme_text})FIGURETHEME\"},\n")
   endforeach()
   string(APPEND FIGURE_THEMES_INC "};\n")

# Same configuration cimgui/cimplot were generated with: their C structs must match the C++ ones bit for bit.
# IMGUI_ENABLE_TEST_ENGINE compiles in Dear ImGui's item hooks, implemented by FIGURE (figure_items.cpp: no
# imgui_test_engine code). It changes no structure (ImGuiContext has its TestEngineHookItems either way) and adds only
# enum values; with the hooks off (g.TestEngineHookItems false: every program that does not record items) a registered
# item costs one test of that boolean.
target_compile_definitions(figure_native PUBLIC
   IMGUI_USER_CONFIG="${cimgui_SOURCE_DIR}/cimconfig.h"
   IMGUI_DISABLE_OBSOLETE_FUNCTIONS=1
   IMGUI_ENABLE_TEST_ENGINE)
# No contraction of a * b + c into a fused multiply-add (GCC and Clang contract by default where the target has FMA:
# aarch64, x86-64 with -march=haswell and later): Dear ImGui's and ImPlot's layout arithmetic and the CPU renderer's
# then round the same way on every platform, so a picture rendered headless does not depend on the machine (#7 S1: an
# FMA build moved an axis by one pixel row). MSVC does not contract by default. scripts/fpm_package.py sets the same
# with pragmas in imconfig.h.
target_compile_options(figure_native PRIVATE
   $<$<COMPILE_LANG_AND_ID:CXX,GNU,Clang,AppleClang,IntelLLVM>:-ffp-contract=off>
   $<$<COMPILE_LANG_AND_ID:C,GNU,Clang,AppleClang,IntelLLVM>:-ffp-contract=off>)
target_include_directories(figure_native PUBLIC
   ${CMAKE_CURRENT_SOURCE_DIR} ${cimgui_SOURCE_DIR} ${IMGUI_DIR} ${IMGUI_DIR}/backends ${cimplot_SOURCE_DIR} ${IMPLOT_DIR}
   ${cimplot3d_SOURCE_DIR} ${IMPLOT3D_DIR})

# Apple: SDL's Objective-C uses clang builtins (___isPlatformVersionAtLeast, behind @available) from clang's runtime
# library, which clang's driver links by itself and gfortran's does not. Installed as libfigure_clang_rt.a for the
# macos varset of fobos (the path of the original depends on the Xcode version).
if(APPLE)
   execute_process(COMMAND ${CMAKE_C_COMPILER} -print-resource-dir OUTPUT_VARIABLE _clang_resource_dir
                   OUTPUT_STRIP_TRAILING_WHITESPACE)
   set(_clang_rt "${_clang_resource_dir}/lib/darwin/libclang_rt.osx.a")
   if(NOT EXISTS "${_clang_rt}")
      message(FATAL_ERROR "clang runtime library not found: ${_clang_rt} (build the native layer with Apple clang)")
   endif()
   install(FILES "${_clang_rt}" DESTINATION lib RENAME libfigure_clang_rt.a)
   # fpm.toml links figure_platform last. With GNU ld (Linux, Windows) scripts/check_sync.py --fpm-link writes it as
   # a linker script adding the platform libraries after fpm's objects; ld64 reads no linker scripts and does not
   # depend on link order, so on macOS it is a real (near-empty) archive and the libraries go in fpm's --link-flag.
   file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/figure_platform.c"
        "/* placeholder of fpm's figure_platform on macOS: see native/CMakeLists.txt */\nint figure_platform_placeholder;\n")
   add_library(figure_platform STATIC "${CMAKE_CURRENT_BINARY_DIR}/figure_platform.c")
