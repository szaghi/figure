# FIGURE native layer: Dear ImGui + ImPlot + cimgui + cimplot + SDL3/OpenGL3 backends + FIGURE glue.
#
# Built once, before any Fortran build, by scripts/build_native.sh. Neither FoBiS nor fpm compiles C++: both only link
# the static libraries this project installs into <prefix>/lib (default: the repository's lib/).
#
# Pinned third-party versions:
#   cimgui  1.93.1WIPdock -> Dear ImGui 1.93.1 WIP, docking branch (v1.93.0-docking + 10 commits). The tag 1.93dock,
#                            despite its name, points to Dear ImGui's master branch and has no docking.
#   cimplot v1.1          -> ImPlot 1.2 WIP (git submodule of cimplot); generated against cimgui 1.93dock, the same
#                            Dear ImGui 1.93 API (docking only adds to it)
#   SDL     release-3.4.20
#   miniz   3.0.2         -> deflate for the remote mode's compressed messages (MIT)
#   stb_image_write v1.16 -> JPEG for the remote mode's changing images (MIT or public domain), vendored unmodified in
#                            third_party/stb (one header)
cmake_minimum_required(VERSION 3.24)
project(figure_native LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

include(FetchContent)

set(FIGURE_CIMGUI_TAG  "1.93.1WIPdock"  CACHE STRING "cimgui git tag")
set(FIGURE_CIMPLOT_TAG "v1.1"           CACHE STRING "cimplot git tag")
set(FIGURE_SDL_TAG     "release-3.4.20" CACHE STRING "SDL git tag")

# ---- SDL3, static ---------------------------------------------------------------------------------------------------
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
# Installed next to the native layer: lib/pkgconfig/sdl3.pc carries the exact static link line of this platform
# (scripts/fpm.sh reads it; scripts/check_sync.py checks the fobos varset of this platform against it).
set(SDL_INSTALL ON CACHE BOOL "" FORCE)
set(SDL_INSTALL_DOCS OFF CACHE BOOL "" FORCE)
# Only what FIGURE uses (window, events, OpenGL, file dialogs): every other subsystem adds link dependencies
# (ALSA/PulseAudio, CoreAudio, WASAPI, game controllers, ...) on every platform.
foreach(_sdl_off AUDIO GPU RENDER CAMERA JOYSTICK HAPTIC HIDAPI SENSOR POWER TRAY)
   set(SDL_${_sdl_off} OFF CACHE BOOL "" FORCE)
endforeach()
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SDL3
   GIT_REPOSITORY https://github.com/libsdl-org/SDL
   GIT_TAG        ${FIGURE_SDL_TAG}
   GIT_SHALLOW    TRUE)
FetchContent_MakeAvailable(SDL3)

# ---- cimgui (with Dear ImGui) and cimplot (with ImPlot): sources only, compiled below -------------------------------
FetchContent_Declare(cimgui
   GIT_REPOSITORY https://github.com/cimgui/cimgui
   GIT_TAG        ${FIGURE_CIMGUI_TAG}
   GIT_SHALLOW    TRUE
   GIT_SUBMODULES imgui)
FetchContent_Declare(cimplot
   GIT_REPOSITORY https://github.com/cimgui/cimplot
   GIT_TAG        ${FIGURE_CIMPLOT_TAG}
   GIT_SHALLOW    TRUE
   GIT_SUBMODULES implot)
FetchContent_GetProperties(cimgui)
if(NOT cimgui_POPULATED)
   FetchContent_Populate(cimgui)
endif()
FetchContent_GetProperties(cimplot)
if(NOT cimplot_POPULATED)
   FetchContent_Populate(cimplot)
endif()

# ---- miniz: deflate (remote mode), one C file from its release archive -------------------------------------------
FetchContent_Declare(miniz
   URL      https://github.com/richgel999/miniz/releases/download/3.0.2/miniz-3.0.2.zip
   URL_HASH SHA256=ada38db0b703a56d3dd6d57bf84a9c5d664921d870d8fea4db153979fb5332c5)
FetchContent_GetProperties(miniz)
if(NOT miniz_POPULATED)
   FetchContent_Populate(miniz)
endif()

set(IMGUI_DIR  ${cimgui_SOURCE_DIR}/imgui)
set(IMPLOT_DIR ${cimplot_SOURCE_DIR}/implot)

add_library(figure_native STATIC
   ${cimgui_SOURCE_DIR}/cimgui.cpp
   ${IMGUI_DIR}/imgui.cpp
   ${IMGUI_DIR}/imgui_draw.cpp
   ${IMGUI_DIR}/imgui_widgets.cpp
   ${IMGUI_DIR}/imgui_tables.cpp
   ${IMGUI_DIR}/imgui_demo.cpp
   ${IMGUI_DIR}/backends/imgui_impl_sdl3.cpp
   ${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp
   ${cimplot_SOURCE_DIR}/cimplot.cpp
   ${IMPLOT_DIR}/implot.cpp
   ${IMPLOT_DIR}/implot_items.cpp
   ${IMPLOT_DIR}/implot_demo.cpp
   ${CMAKE_CURRENT_SOURCE_DIR}/figure_native.cpp
   ${CMAKE_CURRENT_SOURCE_DIR}/figure_process.cpp
   ${CMAKE_CURRENT_SOURCE_DIR}/figure_task.cpp
   ${CMAKE_CURRENT_SOURCE_DIR}/figure_files.cpp
   ${CMAKE_CURRENT_SOURCE_DIR}/figure_glue_generated.cpp  # generated by scripts/gen_bindings.py
   ${CMAKE_CURRENT_SOURCE_DIR}/figure_style_generated.cpp # generated by scripts/gen_bindings.py
   ${CMAKE_CURRENT_SOURCE_DIR}/figure_theme.cpp
   ${CMAKE_CURRENT_SOURCE_DIR}/figure_colormap.cpp
   ${CMAKE_CURRENT_SOURCE_DIR}/figure_view3d.cpp
   ${CMAKE_CURRENT_SOURCE_DIR}/figure_remote.cpp
   ${miniz_SOURCE_DIR}/miniz.c)
# miniz: only its deflate and zlib-format functions, all under the mz_ prefix (no zlib names: a program may link zlib)
target_compile_definitions(figure_native PRIVATE MINIZ_NO_ARCHIVE_APIS MINIZ_NO_STDIO MINIZ_NO_TIME
                                                 MINIZ_NO_ZLIB_COMPATIBLE_NAMES)
target_include_directories(figure_native PRIVATE ${miniz_SOURCE_DIR})
# stb_image_write: the JPEG encoder of figure_remote.cpp (implementation compiled there, its functions static)
target_include_directories(figure_native PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/stb)
# The browser client page, embedded in figure_remote.cpp as a C++ raw string (reconfigured when the page changes).
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/figure_remote_client.html)
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/figure_remote_client.html FIGURE_REMOTE_CLIENT)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/figure_remote_client.inc
     "// generated from native/figure_remote_client.html by native/CMakeLists.txt\n"
     "static const char client_html[] = R\"FIGUREHTML(${FIGURE_REMOTE_CLIENT})FIGUREHTML\";\n")
# The themes, looks and palettes, embedded in figure_theme.cpp as C++ raw strings, one per themes/<name>.toml,
# themes/looks/<name>.toml and themes/palettes/<name>.toml (reconfigured when one changes).
set(FIGURE_THEMES_INC "// generated from themes/*.toml, themes/looks/*.toml and themes/palettes/*.toml by native/CMakeLists.txt\nstruct theme_source { const char *name; const char *text; };\n")
foreach(theme_kind THEMES LOOKS PALETTES)
   if(theme_kind STREQUAL "THEMES")
      set(theme_dir ${CMAKE_CURRENT_SOURCE_DIR}/../themes)
   elseif(theme_kind STREQUAL "LOOKS")
      set(theme_dir ${CMAKE_CURRENT_SOURCE_DIR}/../themes/looks)
   else()
      set(theme_dir ${CMAKE_CURRENT_SOURCE_DIR}/../themes/palettes)
   endif()
   file(GLOB FIGURE_THEME_FILES ${theme_dir}/*.toml)
   list(SORT FIGURE_THEME_FILES)
   set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${FIGURE_THEME_FILES} ${theme_dir})
   string(APPEND FIGURE_THEMES_INC "const theme_source ${theme_kind}[] = {\n")
   foreach(theme_file ${FIGURE_THEME_FILES})
      get_filename_component(theme_name ${theme_file} NAME_WE)
      file(READ ${theme_file} theme_text)
      string(APPEND FIGURE_THEMES_INC "   {\"${theme_name}\", R\"FIGURETHEME(${theme_text})FIGURETHEME\"},\n")
   endforeach()
   string(APPEND FIGURE_THEMES_INC "};\n")
endforeach()
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/figure_themes.inc "${FIGURE_THEMES_INC}")
# The bundled colormaps, embedded in figure_colormap.cpp as C++ raw strings, one per third_party/colormaps/<name>.csv
# (256 RGB rows each, written by scripts/gen_colormaps.py; reconfigured when one changes).
file(GLOB FIGURE_COLORMAP_FILES ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/colormaps/*.csv)
list(SORT FIGURE_COLORMAP_FILES)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${FIGURE_COLORMAP_FILES}
             ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/colormaps)
set(FIGURE_COLORMAPS_INC "// generated from third_party/colormaps/*.csv by native/CMakeLists.txt\nstruct colormap_source { const char *name; const char *text; };\nconst colormap_source COLORMAPS[] = {\n")
foreach(colormap_file ${FIGURE_COLORMAP_FILES})
   get_filename_component(colormap_name ${colormap_file} NAME_WE)
   file(READ ${colormap_file} colormap_text)
   string(APPEND FIGURE_COLORMAPS_INC "   {\"${colormap_name}\", R\"FIGURECMAP(${colormap_text})FIGURECMAP\"},\n")
endforeach()
string(APPEND FIGURE_COLORMAPS_INC "};\n")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/figure_colormaps.inc "${FIGURE_COLORMAPS_INC}")
# The bundled fonts (third_party/fonts), embedded as byte arrays: IBM Plex Sans (OFL-1.1), the default font of the
# interface, IBM Plex Mono (OFL-1.1), the monospace one, and the Lucide icons (ISC), merged into every text font.
set(FIGURE_FONT_FILE ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/fonts/IBMPlexSans-Regular.ttf)
set(FIGURE_MONO_FONT_FILE ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/fonts/IBMPlexMono-Regular.ttf)
set(FIGURE_ICON_FONT_FILE ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/fonts/lucide.ttf)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${FIGURE_FONT_FILE} ${FIGURE_MONO_FONT_FILE}
             ${FIGURE_ICON_FONT_FILE})
file(READ ${FIGURE_FONT_FILE} FIGURE_FONT_HEX HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," FIGURE_FONT_BYTES "${FIGURE_FONT_HEX}")
file(READ ${FIGURE_MONO_FONT_FILE} FIGURE_MONO_FONT_HEX HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," FIGURE_MONO_FONT_BYTES "${FIGURE_MONO_FONT_HEX}")
file(READ ${FIGURE_ICON_FONT_FILE} FIGURE_ICON_FONT_HEX HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," FIGURE_ICON_FONT_BYTES "${FIGURE_ICON_FONT_HEX}")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/figure_font.inc
     "// generated from third_party/fonts/{IBMPlexSans-Regular,IBMPlexMono-Regular,lucide}.ttf by native/CMakeLists.txt\n"
     "static const unsigned char plex_sans_ttf[] = {${FIGURE_FONT_BYTES}};\n"
     "static const unsigned char plex_mono_ttf[] = {${FIGURE_MONO_FONT_BYTES}};\n"
     "static const unsigned char lucide_ttf[] = {${FIGURE_ICON_FONT_BYTES}};\n")
target_include_directories(figure_native PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
if(WIN32)
   target_link_libraries(figure_native PUBLIC ws2_32)
endif()
# Same configuration cimgui/cimplot were generated with: their C structs must match the C++ ones bit for bit.
target_compile_definitions(figure_native PUBLIC
   IMGUI_USER_CONFIG="${cimgui_SOURCE_DIR}/cimconfig.h"
   IMGUI_DISABLE_OBSOLETE_FUNCTIONS=1)
target_include_directories(figure_native PUBLIC
   ${CMAKE_CURRENT_SOURCE_DIR} ${cimgui_SOURCE_DIR} ${IMGUI_DIR} ${IMGUI_DIR}/backends ${cimplot_SOURCE_DIR} ${IMPLOT_DIR})
target_link_libraries(figure_native PUBLIC SDL3::SDL3-static)

install(TARGETS figure_native ARCHIVE DESTINATION lib)
# Apple: SDL's Objective-C uses clang builtins (___isPlatformVersionAtLeast, behind @available) from clang's runtime
# library, which clang's driver links by itself and gfortran's does not. Installed as libfigure_clang_rt.a for the
# macos varset of fobos (the path of the original depends on the Xcode version).
if(APPLE)
   execute_process(COMMAND ${CMAKE_C_COMPILER} -print-resource-dir OUTPUT_VARIABLE _clang_resource_dir
                   OUTPUT_STRIP_TRAILING_WHITESPACE)
   set(_clang_rt "${_clang_resource_dir}/lib/darwin/libclang_rt.osx.a")
   if(NOT EXISTS "${_clang_rt}")
      message(FATAL_ERROR "clang runtime library not found: ${_clang_rt} (build the native layer with Apple clang)")
   endif()
   install(FILES "${_clang_rt}" DESTINATION lib RENAME libfigure_clang_rt.a)
   # fpm.toml links figure_platform last. With GNU ld (Linux, Windows) scripts/check_sync.py --fpm-link writes it as
   # a linker script adding the platform libraries after fpm's objects; ld64 reads no linker scripts and does not
   # depend on link order, so on macOS it is a real (near-empty) archive and the libraries go in fpm's --link-flag.
   file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/figure_platform.c"
        "/* placeholder of fpm's figure_platform on macOS: see native/CMakeLists.txt */\nint figure_platform_placeholder;\n")
   add_library(figure_platform STATIC "${CMAKE_CURRENT_BINARY_DIR}/figure_platform.c")
   install(TARGETS figure_platform ARCHIVE DESTINATION lib)
endif()
# The static SDL3 archive, next to ours, so the Fortran link line is just -lfigure_native -lSDL3.
install(FILES $<TARGET_FILE:SDL3::SDL3-static> DESTINATION lib)
install(FILES ${CMAKE_CURRENT_SOURCE_DIR}/figure_native.h DESTINATION include)

install(FILES "${cimgui_SOURCE_DIR}/imgui/LICENSE.txt" DESTINATION share/licenses/imgui)
install(FILES "${cimplot_SOURCE_DIR}/implot/LICENSE" DESTINATION share/licenses/implot)
install(FILES "${cimgui_SOURCE_DIR}/LICENSE" DESTINATION share/licenses/cimgui)
install(FILES "${cimplot_SOURCE_DIR}/LICENSE" DESTINATION share/licenses/cimplot)
install(FILES "${miniz_SOURCE_DIR}/LICENSE" DESTINATION share/licenses/miniz)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/stb/LICENSE-stb.txt" DESTINATION share/licenses/stb)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../LICENSE.mit.md" DESTINATION share/licenses/figure)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/fonts/LICENSE-IBMPlexSans.txt" DESTINATION share/licenses/ibm-plex)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/fonts/LICENSE-IBMPlexMono.txt" DESTINATION share/licenses/ibm-plex)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/fonts/LICENSE-Lucide.txt" DESTINATION share/licenses/lucide)
# The bundled palettes' licences (themes/palettes: their colours are the upstream projects' own, MIT)
file(GLOB FIGURE_PALETTE_LICENSES ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/palettes/LICENSE-*.txt)
foreach(palette_license ${FIGURE_PALETTE_LICENSES})
   get_filename_component(palette_project ${palette_license} NAME_WE)
   string(REPLACE "LICENSE-" "" palette_project ${palette_project})
   install(FILES ${palette_license} DESTINATION share/licenses/${palette_project})
endforeach()
# The bundled colormaps' licences (third_party/colormaps/LICENSE-<project>.txt, see THIRD_PARTY.md)
file(GLOB FIGURE_COLORMAP_LICENSES ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/colormaps/LICENSE-*.txt)
foreach(colormap_license ${FIGURE_COLORMAP_LICENSES})
   get_filename_component(colormap_project ${colormap_license} NAME_WE)
   string(REPLACE "LICENSE-" "" colormap_project ${colormap_project})
   install(FILES ${colormap_license} DESTINATION share/licenses/${colormap_project})
endforeach()
